
\begin{prooftree}

  \RightLabel{\textsc{StopE}}
  \UnaryInfC{ runE $v_0$ (($s_0$, [::]) :: a) $\boxed{s_0, a}$}

\begin{prooftree}

\end{prooftree}

\end{prooftree}
